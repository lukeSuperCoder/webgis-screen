## 3. API接口设计

### 3.1 监测数据查询接口

#### 接口定义
```javascript
/**
 * 获取监测井的水质数据
 * @param {Object} params
 * @param {string} params.monitoringWellCode - 监测井编码 (必填)
 * @param {string} params.date - 时间 (可选, ISO格式, 默认当前时间)
 * @returns {Promise}
 */
export function getSampleData(params) {
  return request({
    url: '/monitor/sample/data',
    method: 'get',
    params: params
  })
}
```

### 3.2 批量查询策略

由于 `getSampleData()` 只能查询单个监测井的数据，需要实现批量查询策略：

```javascript
/**
 * 批量获取多个监测井的水质数据
 * @param {Array<string>} wellCodes - 监测井编码数组
 * @param {string} date - 查询日期 (可选)
 * @returns {Promise<Array>}
 */
async function batchGetSampleData(wellCodes, date = null) {
  const promises = wellCodes.map(wellCode => {
    return getSampleData({
      monitoringWellCode: wellCode,
      date: date
    }).catch(err => {
      console.warn(`获取监测井 ${wellCode} 数据失败:`, err)
      return null  // 失败时返回null，不影响其他请求
    })
  })

  const results = await Promise.all(promises)
  // 过滤掉失败的请求
  return results.filter(result => result !== null && result.code === 200)
}
```

```javascript
/**
 * 加载综合水质分布数据
 */
async loadComprehensiveWaterQuality() {
  try {
    this.isLoadingSampleData = true

    // 1. 先获取监测井列表
    if (this.wellData.length === 0) {
      await this.loadWellData()
    }

    // 2. 提取监测井编码
    const wellCodes = this.wellData.map(well => well.wellCode)

    if (wellCodes.length === 0) {
      this.$message.warning('暂无监测井数据')
      return
    }

    // 3. 批量查询水质数据
    const results = await this.batchGetSampleData(wellCodes, this.queryDate)

    // 4. 将水质数据映射到 sampleData
    this.sampleData = {}
    results.forEach(result => {
      if (result.data) {
        this.sampleData[result.data.monitoringWellCode] = result.data
      }
    })

    // 5. 生成地图标记数据
    const mapData = this.generateComprehensiveMapData()

    // 6. 渲染到地图
    this.renderComprehensiveLayer(mapData)

    // 7. 显示图例
    this.showComprehensiveLegend()

    this.$message.success(`成功加载 ${results.length} 个监测井的水质数据`)

  } catch (error) {
    console.error('加载综合水质数据失败:', error)
    this.$message.error('加载水质数据失败，请重试')
  } finally {
    this.isLoadingSampleData = false
  }
}

/**
 * 批量获取水质数据
 */
async batchGetSampleData(wellCodes, date = null) {
  const promises = wellCodes.map(wellCode => {

/**
 * 生成综合水质地图数据
 */
generateComprehensiveMapData() {
  const mapData = []

  this.wellData.forEach(well => {
    const sampleData = this.sampleData[well.wellCode]

    if (!sampleData) {
      // 没有水质数据的井，显示为灰色
      mapData.push({
        coordinates: [well.longitude, well.latitude],
        properties: {
          popupType: 'comprehensive',
          wellCode: well.wellCode,
          wellName: well.wellCode,
          measureTime: '暂无数据',
          overallClass: '未知',
          color: '#999999',
          metrics: {}
        }
      })
      return
    }

    // 构建指标对象
    const metrics = {}
    if (sampleData.metrics && Array.isArray(sampleData.metrics)) {
      sampleData.metrics.forEach(metric => {
        // 将指标数据映射为前端所需格式
        const key = this.getMetricKey(metric.metricCode)
        metrics[key] = {
          value: metric.value + (metric.unit || ''),
          class: metric.qualityLevel || ''
        }
      })
    }

    // 确定水质等级颜色
    const color = this.getClassColor(sampleData.qualityLevel || '未知')

    mapData.push({
      coordinates: [well.longitude, well.latitude],
      properties: {
        popupType: 'comprehensive',
        projectName: well.projectId || '',
        wellCode: well.wellCode,
        measureTime: sampleData.samplingTime ? this.formatDateTime(sampleData.samplingTime) : '未知',
        overallClass: sampleData.qualityLevel || '未知',
        color: color,
        metrics: metrics
      }
    })
  })

  return mapData
}

/**
 * 将后端指标编码映射为前端字段名
 */
getMetricKey(metricCode) {
  const mapping = {
    'WATER_TEMP': 'waterTemp',
    'TURBIDITY': 'turbidity',
    'PH': 'ph',
    'DO': 'dissolvedOxygen',
    'CONDUCTIVITY': 'conductivity',
    'CHLOROPHYLL_A': 'chlorophyllA',
    'CYANOBACTERIA': 'cyanobacteria',
    'COD_MN': 'permanganateIndex',
    'TP': 'totalPhosphorus',
    'NH3_N': 'ammoniaNitrogen',
    'TN': 'totalNitrogen',
    'TOTAL_IRON': 'totalIron'
  }
  return mapping[metricCode] || metricCode.toLowerCase()
}

/**
 * 渲染综合水质图层
 */
renderComprehensiveLayer(mapData) {
  // 清除现有标记
  if (this.mapInstance && this.mapInstance.markerLayer) {
    this.mapInstance.markerLayer.clearMarkers()
  }

  // 添加新标记
  mapData.forEach(item => {
    this.mapInstance.markerLayer.addMarker({
      coordinate: item.coordinates,
      shapeType: 21,  // 圆形
      style: {
        fill: item.properties.color,
        stroke: '#fff',
        strokeWidth: 2,
        size: 16
      },
      data: item.properties,
      onClick: (featureData) => {
        this.handleMarkerClick(featureData)
      }
    })
  })

  // 自适应视图
  if (mapData.length > 0) {
    const coordinates = mapData.map(item => item.coordinates)
    this.mapInstance.view.fitExtent(coordinates)
  }
}

/**
```
```javascript
 * 加载单项水质分布数据
 * @param {string} metricName - 指标名称 (pH, 总磷等)
 */
async loadSingleMetricData(metricName) {
  try {
    this.isLoadingSampleData = true

    // 1. 先获取监测井列表
    if (this.wellData.length === 0) {
      await this.loadWellData()
    }

    // 2. 提取监测井编码
    const wellCodes = this.wellData.map(well => well.wellCode)

    if (wellCodes.length === 0) {
      this.$message.warning('暂无监测井数据')
      return
    }

    // 3. 批量查询水质数据
    const results = await this.batchGetSampleData(wellCodes, this.queryDate)

    // 4. 提取指标数据
    const metricData = this.extractMetricData(results, metricName)

    // 5. 渲染到地图
    this.renderSingleMetricLayer(metricData, metricName)

    // 6. 显示图例
    this.showSingleMetricLegend(metricName)

    this.$message.success(`成功加载 ${metricData.length} 个监测井的${metricName}数据`)

  } catch (error) {
    console.error(`加载${metricName}数据失败:`, error)
    this.$message.error(`加载${metricName}数据失败，请重试`)
  } finally {
    this.isLoadingSampleData = false
  }
}

/**
 * 提取指标数据
 */
extractMetricData(results, metricName) {
  const metricCodeMap = {
    'pH': 'PH',
    '总磷': 'TP',
    '氨氮': 'NH3_N',
    '溶解氧': 'DO',
    '高锰酸盐指数': 'COD_MN'
  }

  const metricCode = metricCodeMap[metricName] || metricName
  const mapData = []

  results.forEach(result => {
    if (!result.data) return

    const sampleData = result.data
    const wellCode = sampleData.monitoringWellCode

    // 查找对应的监测井位置
    const well = this.wellData.find(w => w.wellCode === wellCode)
    if (!well) return

    // 查找指标值
    const metric = sampleData.metrics.find(m => m.metricCode === metricCode)

    if (!metric) {
      // 没有该指标数据
      mapData.push({
        coordinates: [well.longitude, well.latitude],
        properties: {
          popupType: 'singleItem',
          wellCode: wellCode,
          metricName: metricName,
          value: null,
          unit: '',
          qualityLevel: '未知',
          color: '#999999'
        }
      })
      return
    }

    // 确定颜色（根据质量等级）
    const color = this.getClassColor(metric.qualityLevel || '未知')

    mapData.push({
      coordinates: [well.longitude, well.latitude],
      properties: {
        popupType: 'singleItem',
        wellCode: wellCode,
        metricName: metricName,
        value: metric.value,
        unit: metric.unit || '',
        qualityLevel: metric.qualityLevel || '未知',
        standardRange: metric.standardRange || '',
        samplingTime: sampleData.samplingTime,
        color: color
      }
    })
  })

  return mapData
}
```

```javascript
methods: {
  /**
   * 带重试的批量查询
   */
  async batchGetSampleDataWithRetry(wellCodes, date = null, retries = 2) {
    for (let i = 0; i <= retries; i++) {
      try {
        const results = await this.batchGetSampleData(wellCodes, date)

        // 检查成功率
        const successRate = results.length / wellCodes.length
        if (successRate < 0.5 && i < retries) {
          // 成功率低于50%，且还有重试次数，则重试
          console.warn(`数据获取成功率过低 (${(successRate * 100).toFixed(0)}%), 正在重试...`)
          await new Promise(resolve => setTimeout(resolve, 1000)) // 等待1秒
          continue
        }

        return results

      } catch (error) {
        if (i === retries) {
          throw error
        }
        console.warn(`批量查询失败，正在重试 (${i + 1}/${retries})...`)
        await new Promise(resolve => setTimeout(resolve, 1000))
      }
    }
  }
}
```

## 6. 可能遇到的问题及解决方案

### 问题1: 批量请求耗时过长
**现象**: 50个监测井，每个请求200ms，总共需要10秒

**解决方案**:
1. **并发控制**: 使用 `Promise.all()` 并发请求（已实现）
2. **请求分批**: 分批加载，先显示部分数据
   ```javascript
   async batchGetSampleDataInBatches(wellCodes, batchSize = 10) {
     const results = []
     for (let i = 0; i < wellCodes.length; i += batchSize) {
       const batch = wellCodes.slice(i, i + batchSize)
       const batchResults = await this.batchGetSampleData(batch)
       results.push(...batchResults)
       // 每批次后更新地图
       this.renderPartialData(results)
     }
     return results
   }
   ```
3. **后端优化**: 建议后端提供批量查询接口
   ```
   POST /monitor/sample/data/batch
   Body: {
     wellCodes: ["WELL-1", "WELL-2", ...],
     date: "2025-10-20T17:48:00"
   }
   ```

### 问题2: 数据格式不匹配
**现象**: 后端返回的指标编码与前端不一致

**解决方案**:
- 维护映射表 `getMetricKey()`
- 与后端协商统一编码规范

### 问题3: 时间格式问题
**现象**: 日期选择器选择的时间与后端期望格式不同

**解决方案**:
```javascript
// 统一时间格式转换函数
formatDateForApi(date) {
  if (!date) return null
  // 转换为ISO 8601格式
  return new Date(date).toISOString()
}
```

### 问题4: 地图性能问题
**现象**: 大量标记导致地图卡顿

**解决方案**:
1. 使用聚类图层（已有实现）
   ```javascript
   this.mapInstance.clusterMakerLayer.addClusterMarkers(mapData)
   ```
2. 根据缩放级别动态加载
3. 使用Canvas渲染替代SVG

### 7.1 数据缓存
```javascript
// 使用localStorage缓存水质数据
const CACHE_KEY = 'water_quality_data'
const CACHE_DURATION = 5 * 60 * 1000 // 5分钟

function getCachedData() {
  const cached = localStorage.getItem(CACHE_KEY)
  if (!cached) return null

  const { data, timestamp } = JSON.parse(cached)
  if (Date.now() - timestamp > CACHE_DURATION) {
    localStorage.removeItem(CACHE_KEY)
    return null
  }

  return data
}

function setCachedData(data) {
  localStorage.setItem(CACHE_KEY, JSON.stringify({
    data: data,
    timestamp: Date.now()
  }))
}
```
