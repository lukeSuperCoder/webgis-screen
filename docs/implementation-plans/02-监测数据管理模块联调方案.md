## 3. 方案选择

根据后端接口情况，有以下三种方案：

### 方案A: 后端提供新增/编辑接口（推荐）
**适用场景**: 后端已实现或将实现单条数据的新增/编辑接口

**优点**:
- 用户体验好，可以在页面直接新增/编辑
- 数据验证更灵活
- 支持单条记录修改

**实施步骤**: 见第4节

---

### 方案B: 仅支持批量导入
**适用场景**: 后端只提供Excel导入功能，不支持单条CRUD

**优点**:
- 后端无需额外开发
- 适合批量数据处理

**缺点**:
- 用户体验较差
- 无法快速修正单条错误数据

**实施步骤**:
1. 移除新增/编辑按钮和表单
2. 移除 `addSampleData`, `updateSampleData` 的引用
3. 保留导入功能
4. 添加Excel模板下载功能

---

### 方案C: 前端临时mock（不推荐，仅开发调试用）
**适用场景**: 后端接口未就绪，前端需要继续开发

**实施步骤**:
1. 使用 `localStorage` 模拟数据存储
2. 前端实现CRUD逻辑
3. 后续替换为真实API

**注意**: 此方案仅用于开发阶段，生产环境必须对接真实API

```javascript
import request from './request'

/**
 * 获取监测样本列表
 */
export function getSampleList(params) {
  return request({
    url: '/monitor/sample/list',
    method: 'get',
    params: params
  })
}

/**
 * 导入监测样本数据
 */
export function importSampleData(formData) {
  return request({
    url: '/monitor/sample/import',
    method: 'post',
    data: formData,
    headers: {
      'Content-Type': 'multipart/form-data'
    }
  })
}

/**
 * 新增监测样本数据
 * @param {Object} data - 监测数据对象
 * @param {string} data.monitoringWellCode - 监测井编号
 * @param {string} data.samplingTime - 采样时间 (ISO格式)
 * @param {string} data.qualityLevel - 质量等级
 * @param {Array} data.metricValues - 指标数组
 */
export function addSampleData(data) {
  return request({
    url: '/monitor/sample',
    method: 'post',
    data: data
  })
}

/**
 * 更新监测样本数据
 * @param {Object} data - 监测数据对象
 * @param {number} data.id - 数据ID
 */
export function updateSampleData(data) {
  return request({
    url: '/monitor/sample',
    method: 'put',
    data: data
  })
}

/**
 * 删除监测样本数据
 * @param {string} ids - 数据ID（逗号分隔）
 */
export function deleteSampleData(ids) {
  return request({
    url: `/monitor/sample/${ids}`,
    method: 'delete'
  })
}

/**
 * 导出监测样本数据
 * @param {Object} params - 查询参数
 */
export function exportSampleData(params) {
  return request({
    url: '/monitor/sample/export',
    method: 'post',
    params: params,
    responseType: 'blob'
  })
}

/**
 * 获取监测井的水质数据
 */
export function getSampleData(params) {
  return request({
    url: '/monitor/sample/data',
    method: 'get',
    params: params
  })
}

/**
 * 获取监测数据范围
 */
export function getSampleDataRange(params) {
  return request({
    url: '/monitor/sample/data/range',
    method: 'get',
    params: params
  })
}

/**
 * 获取所有水质指标
 */
export function getSampleMetrics(params) {
  return request({
    url: '/monitor/sample/metrics',
```

```javascript
/**
 * 添加指标
 */
addMetric() {
  this.form.metricValues.push({
    metricCode: '',
    metricName: '',
    value: null,
    unit: '',
    samplingTime: this.form.samplingTime
  })
},

/**
 * 删除指标
 */
removeMetric(index) {
  this.form.metricValues.splice(index, 1)
},

/**
 * 指标选择变化
 */
handleMetricChange(index) {
  const metricValue = this.form.metricValues[index]
  const metric = this.availableMetrics.find(m => m.metricCode === metricValue.metricCode)

  if (metric) {
    metricValue.metricName = metric.metricName
    metricValue.unit = metric.unit || ''
  }
}
```

#### 3.4 保存数据
```javascript
/**
 * 保存数据
 */
async saveData() {
  try {
    // 表单验证
    await this.$refs.dataForm.validate()

    // 验证至少添加一个指标
    if (this.form.metricValues.length === 0) {
      this.$message.warning('请至少添加一个监测指标')
      return
    }

    // 验证指标数据完整性
    for (let i = 0; i < this.form.metricValues.length; i++) {
      const metric = this.form.metricValues[i]
      if (!metric.metricCode || metric.value === null || metric.value === '') {
        this.$message.warning(`第${i + 1}个指标数据不完整`)
        return
      }
    }

    this.saveLoading = true

    // 准备提交数据
    const submitData = {
      ...this.form,
      // 确保采样时间格式正确
      samplingTime: this.formatDateForApi(this.form.samplingTime),
      // 同步采样时间到所有指标
      metricValues: this.form.metricValues.map(metric => ({
        ...metric,
        samplingTime: this.formatDateForApi(this.form.samplingTime)
      }))
    }

    // 调用API
    let response
    if (this.isEdit) {
      response = await updateSampleData(submitData)
    } else {
      response = await addSampleData(submitData)
    }

    if (response.code === 200) {
      this.$message.success(this.isEdit ? '编辑成功' : '新增成功')
      this.dialogVisible = false
      this.loadData()  // 刷新列表
    } else {
      this.$message.error(response.msg || '操作失败')
    }

  } catch (error) {
    if (error !== false) {  // 排除表单验证错误
      console.error('保存数据失败:', error)
      this.$message.error('操作失败，请重试')
    }
  } finally {
    this.saveLoading = false
  }
},

/**
 * 格式化日期为API格式
 */
formatDateForApi(date) {
  if (!date) return null
  return new Date(date).toISOString()
}
```

```javascript
/**
 * 删除单条数据
 */
async handleDelete(row) {
  try {
    await this.$confirm('确认删除该监测数据吗？', '提示', {
      type: 'warning'
    })

    const response = await deleteSampleData(row.id)

    if (response.code === 200) {
      this.$message.success('删除成功')
      this.loadData()  // 刷新列表
    } else {
      this.$message.error(response.msg || '删除失败')
    }

  } catch (error) {
    if (error !== 'cancel') {
      console.error('删除失败:', error)
      this.$message.error('删除失败，请重试')
    }
  }
},

/**
 * 批量删除
 */
async handleBatchDelete() {
  if (this.selectedRows.length === 0) {
    this.$message.warning('请选择要删除的数据')
    return
  }

  try {
    await this.$confirm(`确认删除选中的 ${this.selectedRows.length} 条数据吗？`, '提示', {
      type: 'warning'
    })

    const ids = this.selectedRows.map(row => row.id).join(',')
    const response = await deleteSampleData(ids)

    if (response.code === 200) {
      this.$message.success('删除成功')
      this.selectedRows = []
      this.loadData()
    } else {
      this.$message.error(response.msg || '删除失败')
    }

  } catch (error) {
    if (error !== 'cancel') {
      console.error('批量删除失败:', error)
      this.$message.error('删除失败，请重试')
    }
  }
}
```

```javascript
  // 下载文件
  XLSX.writeFile(workbook, '监测数据导入模板.xlsx')
}
```
